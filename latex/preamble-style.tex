% ---------- the author's recurring "handwriting" devices ----------
% green "Algorithm:" heading
\newcommand{\Algo}[1][Algorithm]{\par\medskip\noindent{\color{inkgreen}\Large\itshape #1:}\par\smallskip}
% purple "# Step n:" heading
\newcommand{\Step}[2][]{\par\medskip\noindent{\color{inkpurple}\sffamily\bfseries\#\,Step~#2:}\ifx&#1&\else\ {\color{inkpurple}\itshape #1}\fi\par\nobreak\smallskip}
% purple questions the author asks the reader (in this notebook questions are purple)
\newcommand{\Q}[1]{\par\medskip\noindent{\color{inkpurple}\itshape #1}\par\smallskip}
% green "Def n:" / "Th n:" headings, optional gold motto
\newcommand{\Def}[2][]{\par\medskip\noindent{\color{inkgreen}\large\itshape #2:}\ifx&#1&\else\ {\color{inkgold}\large\itshape ``#1''}\fi\par\nobreak\smallskip}
% blue "Demo:" heading
\newcommand{\Demo}[1][Demo]{\par\smallskip\noindent{\color{inkblue}\itshape #1:}\par\nobreak\smallskip}
% blue remark heading (in this notebook remarks are blue)
\newcommand{\Rem}[1][Remark]{\par\medskip\noindent{\color{inkblue}\itshape #1:}\ }
% end of a proof
\newcommand{\qedmark}{\ifmmode\qquad\text{\color{inkgold}$\blacksquare$}\else\hfill{\color{inkgold}$\blacksquare$}\fi}
% red subheading inside a chapter (Problem / Solution etc. in green)
\newcommand{\Prob}[1][Problem]{\par\medskip\noindent{\color{inkgreen}\itshape #1:}\par\nobreak\smallskip}
% coloured inline emphasis

\newcommand{\hlr}[1]{{\color{inkred}#1}}
